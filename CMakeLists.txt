cmake_minimum_required(VERSION 3.25)
project(ReplayCapture VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "ReplayCapture requires Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /EHsc /permissive-)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_library(replay_core src/settings.cpp src/buffer.cpp src/media.cpp src/engine.cpp)
target_include_directories(replay_core PUBLIC src)
target_link_libraries(replay_core PUBLIC windowsapp d3d11 dxgi mfplat mfreadwrite mfuuid mf wmcodecdspuuid ole32 oleaut32 propsys avrt)
add_executable(ReplayCapture WIN32 src/main.cpp src/app.rc src/app.manifest)
target_link_libraries(ReplayCapture PRIVATE replay_core comctl32 shell32 comdlg32 shlwapi)
target_link_options(ReplayCapture PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
add_executable(replay_tests tests/tests.cpp)
target_link_libraries(replay_tests PRIVATE replay_core)
add_executable(replay_integration tests/integration.cpp)
target_link_libraries(replay_integration PRIVATE replay_core winmm)
add_executable(replay_decode tests/decode.cpp)
target_link_libraries(replay_decode PRIVATE replay_core)
enable_testing()
add_test(NAME replay_unit COMMAND replay_tests)
install(TARGETS ReplayCapture RUNTIME DESTINATION .)
install(FILES README.md WINDOWS_REPLAY_RECORDER_PLAN.ko.md DESTINATION .)
install(FILES assets/replaycapture-icon.png DESTINATION assets)
install(FILES docs/README.md docs/gui-and-hotkeys.md docs/troubleshooting.md docs/architecture.md docs/validation-report.md DESTINATION docs)
install(FILES docs/GUI_DESIGN_IMPROVEMENT_PLAN.ko.md docs/GUI_IMPLEMENTATION_REPORT.ko.md DESTINATION docs)
install(FILES docs/images/gui-dashboard.png docs/images/gui-settings.png DESTINATION docs/images)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "ReplayCapture-${PROJECT_VERSION}-windows-x64")
include(CPack)
